#include <Arduino.h>
#include <TFT_eSPI.h> // TFT ekranı kontrol etmek için ana kütüphanemiz.

TFT_eSPI tft = TFT_eSPI();  // Kütüphaneden bir ekran nesnesi oluşturuyoruz.

// Test deseni çizim fonksiyonu
void drawTestPattern(int rotation) {
  int screenWidth = tft.width();
  int screenHeight = tft.height();

  Serial.printf("\n--- ROTASYON %d TESTI ---\n", rotation);
  Serial.printf("Ekran Genisligi: %d | Ekran Yuksekligi: %d\n", screenWidth, screenHeight);

  tft.fillScreen(TFT_BLACK);
  tft.drawRect(0, 0, screenWidth, screenHeight, TFT_WHITE);

  tft.fillRect(0, 0, 20, 20, TFT_RED);
  tft.setTextColor(TFT_RED, TFT_BLACK); // Arka plan rengini de belirle
  tft.setTextSize(2);
  tft.setCursor(25, 5);
  tft.print("0,0");

  tft.fillRect(screenWidth - 20, 0, 20, 20, TFT_GREEN);
  tft.setTextColor(TFT_GREEN, TFT_BLACK);
  tft.setCursor(screenWidth - 100, 5);
  tft.print(String(screenWidth - 1) + ",0");

  tft.fillRect(0, screenHeight - 20, 20, 20, TFT_BLUE);
  tft.setTextColor(TFT_BLUE, TFT_BLACK);
  tft.setCursor(25, screenHeight - 20);
  tft.print("0," + String(screenHeight - 1));
  
  tft.fillRect(screenWidth - 20, screenHeight - 20, 20, 20, TFT_YELLOW);
  tft.setTextColor(TFT_YELLOW, TFT_BLACK);
  tft.setCursor(screenWidth - 120, screenHeight - 20);
  tft.print(String(screenWidth - 1) + "," + String(screenHeight - 1));

  tft.setTextColor(TFT_CYAN, TFT_BLACK);
  tft.setTextSize(3);
  tft.drawCentreString("Rot: " + String(rotation), screenWidth / 2, screenHeight / 2, 1);
}

// Kurulum
void setup() {
  Serial.begin(115200);
  Serial.println("\nEkran Testi Baslatiliyor...");
  
  tft.init();
  tft.fillScreen(TFT_BLACK);
}

// Ana Döngü
void loop() {
  tft.setRotation(0);
  drawTestPattern(0);
  delay(5000);

  tft.setRotation(1);
  drawTestPattern(1);
  delay(5000);

  tft.setRotation(2);
  drawTestPattern(2);
  delay(5000);

  tft.setRotation(3);
  drawTestPattern(3);
  delay(5000);
}
